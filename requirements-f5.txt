# Alternative voice cloner: f5_test.py (F5-TTS). Often matches accent/timbre better
# than Chatterbox. MUST live in its OWN venv — it needs torch 2.14, which conflicts with
# the main pipeline's torch 2.6.
#
# Setup:
#   python3.11 -m venv venv_f5
#   source venv_f5/bin/activate
#   pip install -r requirements-f5.txt
#
# f5-tts pulls its own torch/torchaudio, transformers, soundfile, etc. automatically.
#
# Models download automatically from Hugging Face on the first `python f5_test.py` run:
#   - F5TTS_v1_Base weights + vocos vocoder (~1.3 GB)
#   - a Whisper ASR model, used once to transcribe your reference clip
# All cached in ~/.cache/huggingface. Optionally `export HF_TOKEN=...` for faster pulls.

f5-tts==1.1.22
